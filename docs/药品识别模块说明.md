# 药品识别模块开发文档

## 一、模块概述

### 1.1 功能定位
药品识别模块用于帮助老年用户通过拍照识别药品包装上的信息，自动提取药品名称、剂量、用法，并通过语音播报结果。

### 1.2 老年用户痛点
- **拍摄能力弱**：手抖、对焦困难导致照片模糊/倾斜
- **环境光线差**：室内光线不足或反光严重
- **信息提取难**：药品说明文字小，老花眼难以阅读

#### 编写 `image_processor.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
图像预处理模块
功能：调亮、角度校正、减反光
"""

import cv2
import numpy as np
import base64
from PIL import Image
import io

def decode_image(base64_str):
    """解码Base64图片"""
    img_data = base64.b64decode(base64_str.split(',')[1])
    img = Image.open(io.BytesIO(img_data))
    return cv2.cvtColor(np.array(img), cv2.COLOR_RGB2BGR)

def adjust_brightness(image, alpha=1.2, beta=50):
    """
    调亮图片（针对老人拍摄光线不足）
    alpha: 对比度增益（1.0-3.0）
    beta: 亮度增益（0-100）
    """
    return cv2.convertScaleAbs(image, alpha=alpha, beta=beta)

def correct_angle(image, max_angle=15):
    """
    角度校正（±15°范围内）
    检测图片倾斜角度并自动旋转
    """
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    edges = cv2.Canny(gray, 50, 150, apertureSize=3)
    lines = cv2.HoughLines(edges, 1, np.pi/180, 100)
    
    if lines is not None:
        angles = [line[0][1] for line in lines]
        angle = np.median(angles)
        angle = np.degrees(angle) - 90
        
        if abs(angle) <= max_angle:
            h, w = image.shape[:2]
            center = (w // 2, h // 2)
            M = cv2.getRotationMatrix2D(center, angle, 1.0)
            return cv2.warpAffine(image, M, (w, h), 
                                  flags=cv2.INTER_LINEAR, 
                                  borderMode=cv2.BORDER_CONSTANT, 
                                  borderValue=(255, 255, 255))
    
    return image

def reduce_glare(image):
    """
    减反光（使用CLAHE均衡化）
    适配药品包装反光问题
    """
    lab = cv2.cvtColor(image, cv2.COLOR_BGR2LAB)
    l, a, b = cv2.split(lab)
    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
    l = clahe.apply(l)
    return cv2.cvtColor(cv2.merge([l, a, b]), cv2.COLOR_LAB2BGR)

def preprocess_image(base64_image):
    """
    完整预处理流程
    1. 解码图片
    2. 调亮（亮度提升20%）
    3. 角度校正（±15°）
    4. 减反光
    """
    img = decode_image(base64_image)
    img = adjust_brightness(img, alpha=1.2, beta=50)
    img = correct_angle(img, max_angle=15)
    img = reduce_glare(img)
    return img
```

### 3.3 步骤2：对接百度OCR

#### 编写 `ocr_recognizer.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
百度OCR识别模块
功能：识别药品包装文字
"""

from aip import AipOcr
import base64
import re
import sys
import os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))
from config.baidu_config import BAIDU_ASR_CONFIG

# 初始化百度OCR客户端（复用语音API密钥）
client = AipOcr(
    BAIDU_ASR_CONFIG['APP_ID'],
    BAIDU_ASR_CONFIG['API_KEY'],
    BAIDU_ASR_CONFIG['SECRET_KEY']
)

def extract_drug_info(ocr_result):
    """
    从OCR结果提取药品信息
    返回：{name: 药品名称, dosage: 剂量, usage: 用法}
    """
    # 合并所有识别文字
    text = '\n'.join([word['words'] for word in ocr_result['words_result']])
    
    drug_info = {
        'name': '未识别',
        'dosage': '暂无',
        'usage': '暂无'
    }
    
    # 1. 提取药品名称（通常在前3行，包含"片/胶囊/颗粒"等关键词）
    lines = text.split('\n')[:5]
    for line in lines:
        if re.search(r'(片|胶囊|颗粒|注射液|口服液|丸|糖浆)', line):
            # 清洗文字（去除英文、数字、特殊符号）
            name = re.sub(r'[A-Za-z0-9\(\)\[\]（）【】]', '', line).strip()
            if len(name) >= 3:  # 药品名称至少3个字
                drug_info['name'] = name
                break
    
    # 2. 提取剂量（匹配"数字+单位"）
    dosage_match = re.search(r'(\d+\.?\d*)\s*(mg|g|ml|片|粒|袋)', text, re.IGNORECASE)
    if dosage_match:
        drug_info['dosage'] = f"一次{dosage_match.group(0)}"
    
    # 3. 提取用法（匹配"一日/每日/一次"等）
    usage_match = re.search(r'(一日|每日|一次|每次)\s*(\d+)\s*(次|粒|片|袋)', text)
    if usage_match:
        usage_text = usage_match.group(0)
        # 简化表述（适配老人理解）
        usage_text = usage_text.replace('一日', '每天')
        usage_text = usage_text.replace('每日', '每天')
        drug_info['usage'] = usage_text
    
    # 4. 补充常见服用方式
    if '饭后' in text:
        drug_info['usage'] += ', 饭后服用'
    elif '饭前' in text:
        drug_info['usage'] += ', 饭前服用'
    elif '睡前' in text:
        drug_info['usage'] += ', 睡前服用'
    
    return drug_info

def recognize_drug_text(image_cv2):
    """
    调用百度OCR识别图片中的文字
    
    Args:
        image_cv2: OpenCV格式的图片
    
    Returns:
        药品信息字典
    """
    # 将OpenCV图片转为Base64
    _, img_encoded = cv2.imencode('.jpg', image_cv2)
    img_base64 = base64.b64encode(img_encoded).decode('utf-8')
    
    # 调用百度通用文字识别API
    result = client.basicGeneral(img_base64)
    
    if result.get('words_result'):
        return extract_drug_info(result)
    else:
        raise Exception("OCR识别失败，未检测到文字")
```

### 3.4 步骤3：创建模块入口

#### 编写 `drug_recognize/__init__.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""药品识别模块统一入口"""

from .image_processor import preprocess_image
from .ocr_recognizer import recognize_drug_text
import cv2

def process_and_recognize(base64_image):
    """
    完整识别流程
    1. 图像预处理（调亮、校正、减反光）
    2. OCR识别
    3. 结果结构化
    
    Args:
        base64_image: Base64编码的图片
    
    Returns:
        {name: 药品名称, dosage: 剂量, usage: 用法}
    """
    # 步骤1：预处理
    processed_img = preprocess_image(base64_image)
    
    # 步骤2：OCR识别
    drug_info = recognize_drug_text(processed_img)
    
    # 步骤3：简化表述（适配老人理解）
    drug_info['usage'] = simplify_usage(drug_info['usage'])
    
    return drug_info

def simplify_usage(usage_text):
    """
    简化用法表述（适配老人理解）
    """
    replacements = {
        'bid': '每天两次',
        'tid': '每天三次',
        'qd': '每天一次',
        'qn': '睡前',
        'pc': '饭后',
        'ac': '饭前'
    }
    
    for old, new in replacements.items():
        usage_text = usage_text.replace(old, new)
    
    return usage_text
```

## 五、常见问题排查

### Q1: OCR识别准确率低？

**原因**：图像预处理参数不适配

**解决**：
```python

```

### Q2: 药品名称提取错误？

**原因**：正则表达式匹配规则不够精确

**解决**：
```python
# 增强药品名称匹配规则
patterns = [
    r'[\u4e00-\u9fa5]{2,8}(片|胶囊|颗粒)',  # 中文药名
    r'[\u4e00-\u9fa5]{2,8}(注射液|口服液)',  # 液体药物
    r'[\u4e00-\u9fa5]{2,8}(丸|糖浆)',  # 中成药
]

for pattern in patterns:
    match = re.search(pattern, text)
    if match:
        drug_info['name'] = match.group(0)
        break
```

def compress_image(image, max_size=800):
    h, w = image.shape[:2]
    if max(h, w) > max_size:
        scale = max_size / max(h, w)
        image = cv2.resize(image, (int(w*scale), int(h*scale)))
