# 医生推荐模块开发文档

## 一、模块概述

### 1.1 功能定位
医生推荐模块根据老年用户输入的症状，智能推荐合适的医生，并提供距离、出诊时间等信息，帮助老年人快速找到合适的就医选择。

### 1.2 老年用户痛点
- **信息不对称**：不知道该看哪个科室、找哪个医生
- **行动不便**：希望找离家近的医生
- **筛选能力弱**：医生信息复杂，难以比较选择
- **时间不灵活**：需要了解出诊时间，避免白跑

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""MySQL数据库连接"""

import pymysql
from typing import List, Dict

class DoctorDB:
    """医生数据库操作类"""
    
    def __init__(self, host='localhost', user='root', password='', database='medical'):
        """初始化数据库连接"""
        self.conn = pymysql.connect(
            host=host,
            user=user,
            password=password,
            database=database,
            charset='utf8mb4'
        )
        self.cursor = self.conn.cursor(pymysql.cursors.DictCursor)
    
    def query_doctors_by_department(self, departments: List[str]) -> List[Dict]:
        """
        根据科室查询医生
        
        Args:
            departments: 科室列表
        
        Returns:
            医生列表
        """
        if not departments:
            return []
        
        placeholders = ','.join(['%s'] * len(departments))
        sql = f"""
            SELECT id, name, department, hospital, title, specialty,
                   latitude, longitude, address, schedule, rating, patient_count
            FROM doctors
            WHERE department IN ({placeholders})
            ORDER BY rating DESC, patient_count DESC
            LIMIT 50
        """
        
        self.cursor.execute(sql, departments)
        return self.cursor.fetchall()
    
    def query_all_doctors(self, limit=100) -> List[Dict]:
        """查询所有医生（带限制）"""
        sql = """
            SELECT id, name, department, hospital, specialty,
                   latitude, longitude, address, schedule, rating
            FROM doctors
            ORDER BY rating DESC
            LIMIT %s
        """
        self.cursor.execute(sql, (limit,))
        return self.cursor.fetchall()
    
    def close(self):
        """关闭数据库连接"""
        self.cursor.close()
        self.conn.close()
```

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""工具函数"""

import math
from datetime import datetime
import json

def calculate_distance(lat1, lon1, lat2, lon2):
    """
    计算两点距离（Haversine公式）
    
    Args:
        lat1, lon1: 点1的经纬度
        lat2, lon2: 点2的经纬度
    
    Returns:
        距离（公里）
    """
    R = 6371  # 地球半径（公里）
    
    dlat = math.radians(lat2 - lat1)
    dlon = math.radians(lon2 - lon1)
    
    a = (math.sin(dlat / 2) ** 2 +
         math.cos(math.radians(lat1)) * math.cos(math.radians(lat2)) *
         math.sin(dlon / 2) ** 2)
    
    c = 2 * math.asin(math.sqrt(a))
    return R * c

def format_distance(distance_km):
    """
    格式化距离显示
    
    Args:
        distance_km: 距离（公里）

        格式化字符串（如"1.2公里"或"500米"）
    """
    if distance_km < 1:
        return f"{int(distance_km * 1000)}米"
    else:
        return f"{distance_km:.1f}公里"
```

#### 编写 `recommend_engine.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
医生推荐算法核心
多维度推荐：病症匹配度40% + 距离30% + 用户偏好30%
"""

import sys
import os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))
from rag_module.medical_rag_retriever import MedicalRAGRetriever
from .db_connector import DoctorDB
from .utils import calculate_distance, format_distance, parse_schedule

class DoctorRecommendEngine:
    """医生推荐引擎"""
    
    def __init__(self):
        """初始化推荐引擎"""
        self.retriever = MedicalRAGRetriever()
        self.db = DoctorDB()
    
    def recommend(self, symptom, user_location=None, top_k=5):
        """
        多维度推荐医生
        
        Args:
            symptom: 症状描述
            user_location: 用户位置(latitude, longitude)
            top_k: 返回医生数量
        
        Returns:
            推荐医生列表
        """
        # 步骤1：从知识图谱获取症状对应的科室
        departments = self._get_departments_by_symptom(symptom)
        
        # 步骤2：从数据库查询对应科室的医生
        if departments:
            doctors = self.db.query_doctors_by_department(departments)
        else:
            doctors = self.db.query_all_doctors(limit=50)
        
        # 步骤3：计算综合得分
        for doctor in doctors:
            doctor['综合得分'] = self._calculate_score(
                doctor, symptom, user_location
            )
        
        # 步骤4：排序并返回top_k
        doctors.sort(key=lambda x: x['综合得分'], reverse=True)
        top_doctors = doctors[:top_k]
        
        # 步骤5：格式化输出
        return self._format_doctors(top_doctors, user_location)
    
    def _get_departments_by_symptom(self, symptom):
        """
        从知识图谱获取症状对应的科室
        
        TODO: 需在RAG模块中新增方法
        目前简化实现：关键词匹配
        """
        symptom_department_map = {
            '心慌': ['心内科', '心血管内科'],
            '胸闷': ['心内科', '呼吸内科'],
            '头晕': ['神经内科', '心内科'],
            '头痛': ['神经内科', '全科'],
            '发烧': ['全科', '感染科'],
            '咳嗽': ['呼吸内科', '全科'],
            '胃痛': ['消化内科', '全科'],
        }
        
        for keyword, depts in symptom_department_map.items():
            if keyword in symptom:
                return depts
        
        return ['全科']  # 默认推荐全科
    
    def _calculate_score(self, doctor, symptom, user_location):
        """
        计算综合得分
        综合得分 = 病症匹配度(40%) + 距离得分(30%) + 用户偏好(30%)
        """
        # 1. 病症匹配度（40%）
        match_score = self._calculate_match_score(doctor, symptom)
        
        # 2. 距离得分（30%）
        if user_location:
            distance_score = self._calculate_distance_score(
                doctor, user_location
            )
        else:
            distance_score = 50  # 无位置信息时给中等分
        
        # 3. 用户偏好（30%）- 基于评分和接诊量
        preference_score = self._calculate_preference_score(doctor)
        
        # 综合得分
        total_score = (match_score * 0.4 +
                      distance_score * 0.3 +
                      preference_score * 0.3)
        
        return round(total_score, 2)
    
    def _calculate_match_score(self, doctor, symptom):
        """
        计算病症匹配度（0-100分）
        基于医生擅长领域与症状的匹配程度
        """
        specialty = doctor.get('specialty', '')
        
        # 简化实现：关键词匹配
        keywords = ['心慌', '胸闷', '头晕', '头痛', '高血压', '糖尿病']
        match_count = sum(1 for kw in keywords if kw in specialty and kw in symptom)
        
        if match_count > 0:
            return min(100, 70 + match_count * 10)
        else:
            return 50  # 基础分
    
    def _calculate_distance_score(self, doctor, user_location):
        """
        计算距离得分（0-100分）
        距离越近得分越高
        """
        user_lat, user_lon = user_location
        doctor_lat = doctor.get('latitude')
        doctor_lon = doctor.get('longitude')
        
        if not (doctor_lat and doctor_lon):
            return 50  # 无位置信息时给中等分
        
        distance = calculate_distance(user_lat, user_lon, doctor_lat, doctor_lon)
        
        # 距离评分逻辑
        if distance <= 1:
            return 100
        elif distance <= 3:
            return 80
        elif distance <= 5:
            return 60
        elif distance <= 10:
            return 40
        else:
            return 20
    
    def _calculate_preference_score(self, doctor):
        """
        计算用户偏好得分（0-100分）
        基于医生评分和接诊量
        """
        rating = doctor.get('rating', 0)
        patient_count = doctor.get('patient_count', 0)
        
        # 评分得分（占70%）
        rating_score = (rating / 5.0) * 70
        
        # 经验得分（占30%）- 基于接诊量
        if patient_count > 2000:
            experience_score = 30
        elif patient_count > 1000:
            experience_score = 25
        elif patient_count > 500:
            experience_score = 20
        else:
            experience_score = 15
        
        return rating_score + experience_score
    
    def _format_doctors(self, doctors, user_location):
        """
        格式化医生信息输出
        """
        result = []
        
        for doctor in doctors:
            formatted = {
                'name': doctor['name'],
                'department': doctor['department'],
                'hospital': doctor.get('hospital', ''),
                'distance': self._format_distance(doctor, user_location),
                'schedule': parse_schedule(doctor.get('schedule', '')),
                'matchScore': int(doctor['综合得分']),
                'rating': doctor.get('rating', 0)
            }
            result.append(formatted)
        
        return result
    
    def _format_distance(self, doctor, user_location):
        """格式化距离显示"""
        if not user_location:
            return '未知'
        
        user_lat, user_lon = user_location
        doctor_lat = doctor.get('latitude')
        doctor_lon = doctor.get('longitude')
        
        if not (doctor_lat and doctor_lon):
            return '未知'
        
        distance = calculate_distance(user_lat, user_lon, doctor_lat, doctor_lon)
        return format_distance(distance)
    
    def close(self):
        """关闭数据库连接"""
        self.db.close()
```

```python
_engine = None
```
