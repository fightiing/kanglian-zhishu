# 语音识别(ASR)模块开发文档

## 一、模块概述

### 1.1 功能定位
语音识别模块用于将老年用户的语音问题转换为文字，配合RAG+LLM生成答复，实现老年用户的语音咨询功能。

### 1.2 老年用户痛点
- **操作能力不足**：打字困难，语音输入更符合习惯
- **语音识别适配差**：方言、口齿不清导致识别率低
- **环境噪音干扰**：电视声、家人说话等背景噪音
- **语速缓慢**：老年人说话慢，需延长识别时长

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
音频预处理模块
功能：降噪、音量均衡、延长时长
"""

import wave
import numpy as np
from scipy import signal
import io
import base64

def decode_audio(base64_audio):
    """解码Base64音频"""
    audio_data = base64.b64decode(base64_audio.split(',')[1])
    return io.BytesIO(audio_data)

def noise_reduction(audio_data, sample_rate=16000):
    """
    降噪处理（适配老人环境噪音）
    使用高通滤波器去除低频噪音（电视声、空调声）
    
    Args:
        audio_data: 音频数据（numpy数组）
        sample_rate: 采样率
    
    Returns:
        降噪后的音频数据
    """
    # 设计高通滤波器（截止频率200Hz）
    nyquist = sample_rate / 2
    cutoff = 200 / nyquist
    b, a = signal.butter(4, cutoff, btype='high')
    
    # 应用滤波器
    filtered_audio = signal.filtfilt(b, a, audio_data)
    return filtered_audio.astype(np.int16)

def normalize_volume(audio_data):
    """
    音量均衡（适配老人发音小声）
    将音频归一化到合理范围
    """
    max_val = np.max(np.abs(audio_data))
    if max_val > 0:
        # 归一化到80%的最大音量
        normalized = (audio_data / max_val * 32767 * 0.8).astype(np.int16)
        return normalized
    return audio_data

def extend_duration(audio_data, sample_rate=16000, target_duration=10):
    """
    延长识别时长（适配老人语速慢）
    如果音频短于目标时长，补充静音
    
    Args:
        audio_data: 音频数据
        sample_rate: 采样率
        target_duration: 目标时长（秒）
    """
    current_duration = len(audio_data) / sample_rate
    if current_duration < target_duration:
        silence_length = int((target_duration - current_duration) * sample_rate)
        silence = np.zeros(silence_length, dtype=np.int16)
        return np.concatenate([audio_data, silence])
    return audio_data

def preprocess_audio(base64_audio):
    """
    完整预处理流程
    1. 解码音频
    2. 降噪（高通滤波）
    3. 音量均衡
    4. 延长时长（可选）
    
    Args:
        base64_audio: Base64编码的音频
    
    Returns:
        预处理后的音频字节流
    """
    # 解码
    audio_buffer = decode_audio(base64_audio)
    
    # 读取WAV数据
    with wave.open(audio_buffer, 'rb') as wf:
        sample_rate = wf.getframerate()
        audio_data = np.frombuffer(wf.readframes(wf.getnframes()), dtype=np.int16)
    
    # 降噪
    audio_data = noise_reduction(audio_data, sample_rate)
    
    # 音量均衡
    audio_data = normalize_volume(audio_data)
    
    # 转回字节流
    output_buffer = io.BytesIO()
    with wave.open(output_buffer, 'wb') as wf:
        wf.setnchannels(1)
        wf.setsampwidth(2)  # 16-bit
        wf.setframerate(sample_rate)
        wf.writeframes(audio_data.tobytes())
    
    output_buffer.seek(0)
    return output_buffer.read()
```

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
百度ASR接口封装
功能：语音转文字，支持方言识别
"""

from aip import AipSpeech
import base64
import sys
import os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))
from config.baidu_config import BAIDU_ASR_CONFIG

# 初始化百度ASR客户端
client = AipSpeech(
    BAIDU_ASR_CONFIG['APP_ID'],
    BAIDU_ASR_CONFIG['API_KEY'],
    BAIDU_ASR_CONFIG['SECRET_KEY']
)

def recognize_speech(audio_bytes, dialect='mandarin'):
    """
    调用百度ASR识别语音
    
    Args:
        audio_bytes: 音频字节流（WAV格式）
        dialect: 方言类型
            - 'mandarin': 普通话（默认）
            - 'sichuan': 四川话
            - 'cantonese': 粤语
            - 'northeast': 东北话
    
    Returns:
        识别的文字
    """
    # 方言代码映射
    dialect_map = {
        'mandarin': 1537,   # 普通话
        'sichuan': 1737,    # 四川话（需开通方言服务）
        'cantonese': 1637,  # 粤语
        'northeast': 1837   # 东北话
    }
    
    pid = dialect_map.get(dialect, 1537)
    
    # 调用ASR接口
    result = client.asr(
        audio_bytes,
        'wav',
        16000,  # 采样率
        {
            'dev_pid': pid,
            'punctuation': 1,  # 启用标点符号
            'vad': 'touch',     # 语音活动检测（自动识别语音段）
            'confidence': 0.6   # 置信度阈值（降低以适配老人口齿不清）
        }
    )
    
    # 解析结果
    if result.get('err_no') == 0:
        return result['result'][0]
    else:
        raise Exception(f"ASR识别失败: {result.get('err_msg', '未知错误')}")

def auto_detect_dialect(audio_bytes):
    """
    自动检测方言（简化版）
    尝试多种方言识别，返回置信度最高的结果
    """
    dialects = ['mandarin', 'sichuan', 'cantonese']
    
    for dialect in dialects:
        try:
            text = recognize_speech(audio_bytes, dialect=dialect)
            return text, dialect
        except Exception as e:
            continue
    
    raise Exception("所有方言识别均失败")

def recognize_with_retry(audio_bytes, max_retries=3):
    """
    带重试的识别（提高成功率）
    
    Args:
        audio_bytes: 音频字节流
        max_retries: 最大重试次数
    
    Returns:
        识别的文字
    """
    for i in range(max_retries):
        try:
            return recognize_speech(audio_bytes)
        except Exception as e:
            if i == max_retries - 1:
                raise e
            print(f"第{i+1}次识别失败，重试中...")
    
    raise Exception("识别失败，已达最大重试次数")
```

#### 编写 `intent_parser.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
NLP意图解析模块
功能：从用户语音文本中提取疾病关键词和意图
"""

import re
import sys
import os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))

def extract_disease_keywords(text):
    """
    提取疾病关键词（简化版）
    实际应用中可使用NER命名实体识别
    
    Args:
        text: 用户语音转写文本
    
    Returns:
        疾病列表
    """
    # 常见疾病关键词匹配
    disease_patterns = [
        r'高血压', r'糖尿病', r'冠心病', r'心脏病',
        r'感冒', r'发烧', r'咳嗽', r'头痛', r'头晕',
        r'胃病', r'胃炎', r'肠炎', r'胃溃疡',
        r'关节炎', r'风湿', r'痛风',
        r'哮喘', r'支气管炎', r'肺炎'
    ]
    
    diseases = []
    for pattern in disease_patterns:
        if re.search(pattern, text):
            match = re.search(pattern, text).group()
            if match not in diseases:
                diseases.append(match)
    
    return diseases

def parse_intent(text):
    """
    解析用户意图
    
    Args:
        text: 用户语音转写文本
    
    Returns:
        {
            'intent': 'query_disease' | 'query_drug' | 'query_food',
            'entities': ['高血压', ...],
            'query_type': 'symptom' | 'treatment' | 'diet'
        }
    """
    intent_info = {
        'intent': 'query_disease',
        'entities': [],
        'query_type': 'general'
    }
    
    # 提取疾病实体
    diseases = extract_disease_keywords(text)
    intent_info['entities'] = diseases
    
    # 判断查询类型
    if re.search(r'(症状|表现|怎么回事|什么症状)', text):
        intent_info['query_type'] = 'symptom'
    elif re.search(r'(吃什么药|用药|治疗|怎么治)', text):
        intent_info['intent'] = 'query_drug'
        intent_info['query_type'] = 'treatment'
    elif re.search(r'(吃什么|饮食|食物|忌口|不能吃)', text):
        intent_info['intent'] = 'query_food'
        intent_info['query_type'] = 'diet'
    elif re.search(r'(注意什么|需要注意|怎么办)', text):
        intent_info['query_type'] = 'general'
    
    return intent_info

def build_query_from_intent(intent_info):
    """
    根据意图构建结构化查询
    
    Args:
        intent_info: 意图信息
    
    Returns:
        结构化的查询问题
    """
    entities = intent_info['entities']
    query_type = intent_info['query_type']
    
    if not entities:
        return "请问您想咨询哪种疾病?"
    
    disease = entities[0]  # 取第一个疾病
    
    if query_type == 'symptom':
        return f"{disease}有哪些症状表现?"
    elif query_type == 'treatment':
        return f"{disease}患者应该吃什么药?"
    elif query_type == 'diet':
        return f"{disease}患者在饮食上需要注意什么?"
    else:
        return f"{disease}患者需要注意什么?"
```

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""语音询问模块统一入口"""

from .audio_processor import preprocess_audio
from .baidu_asr import recognize_speech
from .intent_parser import parse_intent, build_query_from_intent
import sys
import os
sys.path.append(os.path.join(os.path.dirname(__file__), '..'))
from rag_module.medical_rag_retriever import MedicalRAGRetriever
from llm_module.deepseek_llm import DeepSeekLLM, MedicalAssistant

# 初始化系统组件（单例）
_retriever = None
_llm = None
_assistant = None

def get_components():
    """获取系统组件（懒加载）"""
    global _retriever, _llm, _assistant
    
    if _retriever is None:
        _retriever = MedicalRAGRetriever()
        _llm = DeepSeekLLM(model="deepseek-r1:latest")
        _assistant = MedicalAssistant(_retriever, _llm)
    
    return _retriever, _llm, _assistant

def process_voice_query(base64_audio):
    """
    完整语音询问流程
    1. 音频预处理（降噪、音量均衡）
    2. ASR转写
    3. NLP意图解析
    4. 知识图谱检索
    5. 大模型生成答复
    
    Args:
        base64_audio: Base64编码的音频
    
    Returns:
        {
            'transcript': '转写文本',
            'intent': {...},
            'query': '结构化查询',
            'answer': 'AI答复'
        }
    """
    # 步骤1：音频预处理
    processed_audio = preprocess_audio(base64_audio)
    
    # 步骤2：ASR转写
    transcript = recognize_speech(processed_audio)
    
    # 步骤3：意图解析
    intent = parse_intent(transcript)
    structured_query = build_query_from_intent(intent)
    
    # 步骤4&5：检索知识并生成答复
    retriever, llm, assistant = get_components()
    
    if intent['entities']:
        disease = intent['entities'][0]
        knowledge = retriever.comprehensive_retrieve(disease)
        
        if knowledge:
            prompt = assistant.build_prompt(structured_query, knowledge)
            answer = llm.generate_response(prompt, temperature=0.3)
        else:
            answer = f"抱歉，未找到关于【{disease}】的相关信息。"
    else:
        answer = "抱歉，未能识别您的问题，请重新描述或直接说出疾病名称。"
    
    return {
        'transcript': transcript,
        'intent': intent,
        'query': structured_query,
        'answer': answer
    }
```

### 4.3 测试脚本

```python
# test_voice_ask.py
from voice_ask import process_voice_query
import base64
import os

test_audios = [
    'test_data/clear_voice.wav',
    'test_data/dialect_voice.wav',
    'test_data/noise_voice.wav'
]

for audio_path in test_audios:
    print(f"\n测试音频: {audio_path}")
    
    with open(audio_path, 'rb') as f:
        audio_base64 = 'data:audio/wav;base64,' + base64.b64encode(f.read()).decode()
    
    try:
        result = process_voice_query(audio_base64)
        print(f"✓ 识别成功")
        print(f"  转写文本: {result['transcript']}")
        print(f"  提取实体: {result['intent']['entities']}")
        print(f"  答复: {result['answer'][:50]}...")
    except Exception as e:
        print(f"✗ 处理失败: {e}")
```

## 五、常见问题排查

### Q1: ASR识别准确率低？

**原因分析**：
1. 方言问题
2. 背景噪音过大
3. 录音质量差

**解决方案**：
```python
# 1. 自动检测方言
text, dialect = auto_detect_dialect(audio_bytes)
print(f"检测到方言: {dialect}")

# 2. 增强降噪力度
audio_data = noise_reduction(audio_data, cutoff=300)  # 提高截止频率

# 3. 降低置信度阈值
result = client.asr(audio_bytes, 'wav', 16000, {'confidence': 0.5})
```

### Q2: NLP意图解析失败？

**原因**：疾病关键词库不全

**解决**：
```python
# 扩展疾病关键词库
disease_patterns = [
    # 慢性病
    r'高血压', r'糖尿病', r'冠心病',
    # 常见病
    r'感冒', r'发烧', r'咳嗽', r'头痛',
    # 消化系统
    r'胃病', r'胃炎', r'肠炎', r'胃溃疡',
    # 呼吸系统
    r'哮喘', r'支气管炎', r'肺炎',
    # 骨科
    r'关节炎', r'风湿', r'痛风',
    # 心血管
    r'心脏病', r'心绞痛', r'心律失常'
]
```

### Q3: 百度ASR调用失败？

**检查清单**：
1. API密钥是否正确（`config/baidu_config.py`）
2. 网络连接是否正常
3. 是否超出免费配额（5万次/天）
4. 音频格式是否符合要求（WAV/PCM/MP3，16kHz）

**调试代码**：
```python
# 打印ASR原始结果
result = client.asr(audio_bytes, 'wav', 16000, {})
print("ASR原始结果：", result)

if result.get('err_no') != 0:
    print(f"错误码: {result.get('err_no')}")
    print(f"错误信息: {result.get('err_msg')}")
```

## 六、优化建议

### 6.1 性能优化

1. **组件懒加载**：RAG和LLM只在需要时初始化
2. **音频压缩**：录音前限制时长（最多30秒）
3. **结果缓存**：相同问题缓存答复

### 6.2 准确率优化

1. **方言自动检测**：尝试多种方言，选最优结果
2. **多轮识别**：ASR识别3次，选置信度最高的
3. **上下文记忆**：记录用户最近咨询的疾病
